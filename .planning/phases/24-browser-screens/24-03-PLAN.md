    ```vue
    <script setup lang="ts">
    import { deckMetadata } from '../../src/data/decks'
    import { DECK_MAP } from '../../src/data/deckMap'
    import type { Card, DeckId } from '~/types'

    const { t, locale } = useI18n()
    const lang = computed(() => locale.value as 'it' | 'es')

    const { isCardDueForDeck, hasProgress } = useSRS(lang.value)
    const { activeLevels, setActiveLevels } = useLevelFilter(lang.value, hasProgress)

    const decks = computed(() => deckMetadata.filter(d => d.lang === lang.value))

    function getDueCount(deckId: DeckId, cards: Card[]): number {
      if (!activeLevels.value) return 0
      return cards.filter((card, i) =>
        activeLevels.value!.includes(card.level) && isCardDueForDeck(deckId, i)
      ).length
    }

    const deckDueCounts = computed(() =>
      Object.fromEntries(
        decks.value.map(deck => {
          const cards = DECK_MAP[lang.value]?.[deck.id] ?? []
          return [deck.id, getDueCount(deck.id, cards)]
        })
      )
    )
    </script>
    ```

    TEMPLATE:
    - `<main>` wrapping everything
    - Section header: `<h1>{{ t('page.chooseDeck') }}</h1>`
    - `<div data-level-filter>` wrapping `<LevelFilterChips :active-levels="activeLevels ?? []" @update:active-levels="setActiveLevels" />`
    - `<div class="deck-grid">` with `v-for="deck in decks"` (`:key="deck.id"`)
    - Each tile: `<NuxtLinkLocale :to="'/' + deck.id">` wrapping `<div :class="['deck-card', 'theme-' + deck.theme]">`
    - Inside card: icon circle, `<h3>{{ t('decks.' + deck.i18nKey) }}</h3>`, `<p>{{ t('deckDescriptions.' + deck.i18nKey) }}</p>`
    - Badge: `<span :class="['deck-card-badge', deckDueCounts[deck.id] === 0 ? 'deck-card-badge--done' : '']">{{ deckDueCounts[deck.id] === 0 ? '✓' : deckDueCounts[deck.id] }}</span>`

    REACTIVITY KEY: `deckDueCounts` is a `computed`. It reads `activeLevels.value` and calls `isCardDueForDeck` (which reads `progress.value`). Vue tracks both refs as dependencies. When either changes (chip toggle OR card grade), the computed re-evaluates and the badge updates in the same render cycle.

    NULL GUARD: `getDueCount` returns 0 when `activeLevels.value === null`. This handles the SSR initial state and the brief window between mount and `onMounted` firing in the composable.

    IMPORTANT pitfall to avoid: Do NOT use `activeLevels.value` as a direct prop to `useSRS` or as a static arg to a computed factory. The `activeLevels` ref must be read INSIDE the `computed` callback so Vue tracks it.
