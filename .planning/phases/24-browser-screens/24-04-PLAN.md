    SCRIPT SETUP:
    ```vue
    <script setup lang="ts">
    import { scenarios } from '../../src/data/qa'
    import { isCardDue } from '~/lib/srs'
    import type { Scenario } from '~/types'

    const SCENARIO_THEMES = ['teal', 'blue', 'red', 'yellow', 'green', 'purple', 'orange'] as const

    const { t, locale } = useI18n()
    const lang = computed(() => locale.value as 'it' | 'es')

    const { progress, hasProgress } = useSRS(lang.value)
    const { activeLevels, setActiveLevels } = useLevelFilter(lang.value, hasProgress)

    function getQADueCount(scenario: Scenario): number {
      if (!activeLevels.value) return 0
      return scenario.cards.filter(card =>
        activeLevels.value!.includes(card.level) &&
        isCardDue(progress.value[`qa_${scenario.id}_${card.id}`])
      ).length
    }

    const scenarioDueCounts = computed(() =>
      Object.fromEntries(scenarios.map(s => [s.id, getQADueCount(s)]))
    )
    </script>
    ```

    TEMPLATE:
    - `<main>` wrapping everything
    - Section header: `<h1>{{ t('page.title') }}</h1>` (or appropriate key from messages)
    - `<div data-level-filter>` wrapping `<LevelFilterChips :active-levels="activeLevels ?? []" @update:active-levels="setActiveLevels" />`
    - `<div class="deck-grid">` with `v-for="(scenario, index) in scenarios"` (`:key="scenario.id"`)
    - Each tile: `<NuxtLinkLocale :to="'/qa/' + scenario.id">` wrapping `<div :class="['deck-card', 'theme-' + SCENARIO_THEMES[index % SCENARIO_THEMES.length]]">`
    - Inside card: `<div class="deck-icon-circle">{{ scenario.icon }}</div>`, `<h3>{{ lang === 'it' ? scenario.titleIt : scenario.titleEs }}</h3>`
    - Badge: `<span :class="['deck-card-badge', scenarioDueCounts[scenario.id] === 0 ? 'deck-card-badge--done' : '']">{{ scenarioDueCounts[scenario.id] === 0 ? '✓' : scenarioDueCounts[scenario.id] }}</span>`

    REACTIVITY KEY: `scenarioDueCounts` is a `computed`. It calls `getQADueCount` which reads `activeLevels.value` and `progress.value[...]`. Vue tracks both refs. When a chip is toggled (`setActiveLevels` updates `activeLevels`), the computed re-evaluates and all badge counts update atomically.

    NULL GUARD: `getQADueCount` returns 0 when `activeLevels.value === null`. Prevents SSR crash and handles pre-mount state.

    IMPORTANT pitfall to avoid: Do NOT call `useQASRS()` anywhere in this file — not in setup, not in the scenarios.map(), not in a computed. The `useQASRS` composable takes a plain `Level[]` at call time and loses reactivity to chip toggles. Use `progress` + `isCardDue` directly as shown above.
