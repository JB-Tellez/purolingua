<!-- Types to import in every scenario file -->
From src/types/index.ts (added in Plan 01):
```typescript
import type { QACard, Scenario } from '@/types';

// ScenarioId values: 'caffe' | 'albergo' | 'ristorante' | 'strada' | 'presentazioni' | 'negozio' | 'treno'
// Level values: 'A1' | 'A2'

export interface QACard {
  id: string;                         // "{scenarioId}_{nn}" e.g. "caffe_01"
  question: string;                   // Italian question
  correct: string;                    // Italian correct response
  foils: [string, string, string];    // 3 Italian wrong responses, fixed tuple
  questionEs: string;                 // Spanish question (parallel translation)
  correctEs: string;                  // Spanish correct response (parallel translation)
  foilsEs: [string, string, string];  // 3 Spanish wrong responses, fixed tuple
  level: Level;                       // same as parent Scenario.level
}

export interface Scenario {
  id: ScenarioId;
  icon: string;      // emoji
  titleIt: string;
  titleEs: string;
  level: Level;      // 'A1' or 'A2' — all cards inherit this
  cards: QACard[];
}
```
